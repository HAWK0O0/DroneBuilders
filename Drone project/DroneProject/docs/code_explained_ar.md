/* ============================================================================
 * 1. 程序入口点 (src/main.cpp)
 * ============================================================================
 * 功能：系统初始化与主循环调度。
 */
void setup() {
    // 启动所有子系统：IMU, RC, GPS, Motor
    FlightController::setup(); 
}

void loop() {
    // 执行高频任务：姿态计算与电机输出
    FlightController::loop();
}

/* ============================================================================
 * 2. 全局配置项 (include/controller_config.h)
 * ============================================================================
 * 如何调整：
 * - 飞行响应：增大 MIX_SCALE (默认0.6f) 提升手感。
 * - 自动平稳：增大 AUTO_KP 提升自稳修正速度。
 * - 电机平滑：增大 ESC_STEP 减少电机突发性震动。
 * - 安全距离：修改 FAILSAFE_US (默认300000us) 调整失去信号后的关停时间。
 */

/* ============================================================================
 * 3. 遥控捕获模块 (src/rc_input.cpp)
 * ============================================================================
 * 技术细节：
 * - 硬件中断 (PCINT)：利用端口变化中断实时获取遥控脉冲宽度。
 * - 线程安全：读取 rcUs 数组时会锁定中断，防止主循环读取到一半被修改。
 * - 映射逻辑：将物理引脚 (Pin) 映射至逻辑通道 (Channel)。
 */



/* ============================================================================
 * 4. 传感器管理 (src/imu_manager.cpp)
 * ============================================================================
 * 功能：
 * - BNO055 自动寻址：支持 0x28 和 0x29 地址。
 * - 欧拉角获取：获取 Roll(滚转), Pitch(俯仰), Yaw(偏航) 的绝对度数。
 * - 滤波：应用 IMU_EULER_LPF_ALPHA 进行低通滤波，消除高频振动干扰。
 */

/* ============================================================================
 * 5. 动力混合算法 (src/flight_mixer.cpp)
 * ============================================================================
 * 混合逻辑 (X-Frame Mixer)：
 * - Motor 1 (左前): Throttle - Pitch + Roll - Yaw
 * - Motor 2 (右前): Throttle + Pitch + Roll + Yaw
 * - Motor 3 (右后): Throttle + Pitch - Roll - Yaw
 * - Motor 4 (左后): Throttle - Pitch - Roll + Yaw
 *
 * 处理流程：
 * 1. 将摇杆量转换为角度修正量。
 * 2. 如果开启 Auto-Level，则叠加 IMU 姿态偏差补偿。
 * 3. 应用电机曲线 (applyMotorCurve) 和微调 (Trim)。
 */



/* ============================================================================
 * 6. 底层电机输出 (src/motor_output.cpp)
 * ============================================================================
 * 核心技术：
 * - 零延迟同步：所有电机引脚都在 PORTB。
 * - 定时器 Timer1：产生 0.5us 高精度脉冲。
 * - ISR 同步写入：一次性写入寄存器，确保 4 个电机上升沿完全重合。
 */

/* ============================================================================
 * 7. 综合调度 (src/flight_controller.cpp)
 * ============================================================================
 * 状态机逻辑：
 * - 分支 A：失去信号 -> Failsafe (停机)。
 * - 分支 B：锁定状态 (Disarmed) -> 输出 ESC_STOP (1000us)。
 * - 分支 C：解锁状态 (Armed) -> 根据模式输出混合后的 PWM 信号。
 * - 特殊模式：Passthrough (直通测试) 或 Test Mode (单电机顺序测试)。
 */

/* ============================================================================
 * 调试建议 (Debugging Tips)
 * ============================================================================
 * 1. 螺旋桨安全：在测试 PID 或 映射(Mapping) 时，必须拆卸桨叶！
 * 2. 反向修正：如果自稳模式下飞机翻车，请检查 IMU_*_SIGN 是否需要取反。
 * 3. 油门保护：设置 AUTO_MIN_THR 确保在油门极低时不会胡乱修正。
 */
